open Lean Elab Command
set_option maxRecDepth 10000
set_option maxHeartbeats 0
run_cmd do
  let env ← getEnv
  let wanted : Array Name := TARGETS
  let mut records : Array Json := #[]
  for n in wanted do
    let some c := env.find? n | throwError "missing target {n}"
    let some idx := env.getModuleIdxFor? n | throwError "missing owner {n}"
    let mut used := c.type.getUsedConstants
    if let some v := c.value? true then used := used ++ v.getUsedConstants
    let direct := used.toList.map Name.toString
    let mut todo := #[n]
    let mut seen : NameSet := {}
    let mut deps : Array String := #[]
    while !todo.isEmpty do
      let current := todo.back!
      todo := todo.pop
      if seen.contains current then continue
      seen := seen.insert current
      let some ci := env.find? current | throwError "missing dependency"
      deps := deps.push current.toString
      let mut cs := ci.type.getUsedConstants
      if let some v := ci.value? true then cs := cs ++ v.getUsedConstants
      for d in cs do
        if let some i := env.getModuleIdxFor? d then
          if (env.header.moduleNames[i.toNat]!).toString.startsWith "Aiyagari1994." then
            todo := todo.push d
    let ax ← collectAxioms n
    records := records.push <| Json.mkObj [
      ("name", toJson n.toString),
      ("owner", toJson (env.header.moduleNames[idx.toNat]!).toString),
      ("unsafe", toJson c.isUnsafe), ("partial", toJson c.isPartial),
      ("direct_dependencies", toJson direct.mergeSort),
      ("project_dependencies", toJson deps.toList.mergeSort),
      ("axioms", toJson ((ax.toList.map Name.toString).mergeSort))]
  liftIO <| IO.println <| Json.compress <| toJson records
